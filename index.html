async function wpDetect(){
  const domain=clean(document.getElementById("target").value);
  if(!domain){ el.labOut().textContent="Target required"; return; }
  try{
    const r=await timeoutFetch(PROXY+"http://"+domain,8000);
    const body=await r.text();
    const plugins=[...body.matchAll(/wp-content\/plugins\/([^\/]+)\//gi)].map(x=>x[1]);
    const themes=[...body.matchAll(/wp-content\/themes\/([^\/]+)\//gi)].map(x=>x[1]);
    el.labOut().textContent="WP Detect\nPlugins: "+([...new Set(plugins)].join(", ")||"None")+"\nThemes: "+([...new Set(themes)].join(", ")||"None");
  }catch{ el.labOut().textContent="Failed"; }
}

async function bruteAdmin(resume=false){
  const domain=clean(document.getElementById("target").value);
  if(!domain){ el.adminOut().textContent="Target required"; return; }
  if(!resume){
    bruteState.admin.running=true;
    bruteState.admin.index=0;
    bruteState.admin.found=[];
    bruteState.admin.list=["admin","administrator","admin/login","admin/index","adminpanel","adminarea","wp-admin","wp-login.php","login","dashboard","backend","portal"];
    const file=document.getElementById("adminWordlistFile");
    if(file.files.length){
      const text=await file.files[0].text();
      const lines=text.split("\n").map(x=>x.trim()).filter(Boolean);
      bruteState.admin.list=[...new Set([...bruteState.admin.list,...lines])];
    }
  }
  bruteState.admin.running=true;
  while(bruteState.admin.running && bruteState.admin.indexsetTimeout(r,150));
  }
}

function pauseBrute(type){
  bruteState[type].running=false;
}

function resumeBrute(type){
  if(type==="admin") bruteAdmin(true);
  if(type==="dir") bruteDir(true);
  if(type==="sub") bruteSub(true);
}

function exportReport(){
  if(!lastReport){ alert("No report yet"); return; }
  const blob=new Blob([JSON.stringify(lastReport,null,2)],{type:"application/json"});
  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");
  a.href=url;
  a.download="orca_report.json";
  a.click();
  URL.revokeObjectURL(url);
}
