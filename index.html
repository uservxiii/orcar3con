    async function adminBrute(resume=false){
      const domain = clean(document.getElementById("target").value);
      if(!domain){
        document.getElementById("adminOut").innerHTML="Target required.";
        return;
      }

      if(!resume){
        bruteState.admin.running = true;
        bruteState.admin.index = 0;
        bruteState.admin.found = [];
        bruteState.admin.list = [];
      }

      bruteState.admin.running = true;

      let list = bruteState.admin.list;
      if(!list.length){
        list = ["admin","administrator","admin/login","admin/index","adminpanel","adminarea","cpanel","wp-admin","wp-login.php"];
        bruteState.admin.list = list;
      }

      while(bruteState.admin.running && bruteState.admin.index < list.length){
        const p = list[bruteState.admin.index];
        bruteState.admin.index++;

        try{
          const r = await timeoutFetch(PROXY+"http://"+domain+"/"+p, 4000);
          if(r && r.status === 200){
            bruteState.admin.found.push("/"+p);
          }
        }catch(e){}

        const output =
          `Admin Brute Result:
` +
          `Found: ${bruteState.admin.found.length ? escapeHTML(bruteState.admin.found.join(", ")) : "None"}
` +
          `Checked: ${bruteState.admin.index}/${list.length}`;

        document.getElementById("adminOut").innerHTML = output;
        await new Promise(r=>setTimeout(r, 150));
      }
    }

    // ----------------- CSRF CHECKER -----------------
    function csrfChecker(){
      const domain = clean(document.getElementById("target").value);
      if(!domain){
        document.getElementById("adminOut").innerHTML="Target required.";
        return;
      }

      const body = ""; // karena tidak ada source HTML, ini hanya placeholder
      const forms = [...body.matchAll(//gi)];
      if(!forms.length){
        document.getElementById("adminOut").innerHTML="No forms found on homepage.";
        return;
      }

      let issues = [];
      forms.forEach((f, idx)=>{
        const formHtml = f[0];
        const hasCSRF = /csrf|token|_csrf|csrf_token/i.test(formHtml);
        if(!hasCSRF){
          issues.push(`Form ${idx+1}: MISSING CSRF token`);
        }
      });

      document.getElementById("adminOut").innerHTML =
        `CSRF Token Checker:
` +
        (issues.length ? escapeHTML(issues.join("
")) : "All forms contain CSRF token (or token-like field).") +
        "

(Note: token detection is heuristic-based.)";
    }

    async function subdomainBrute(resume=false){
      const domain = clean(document.getElementById("target").value);
      if(!domain){
        document.getElementById("subOut").innerHTML="Target required.";
        return;
      }

      if(!resume){
        bruteState.sub.running = true;
        bruteState.sub.index = 0;
        bruteState.sub.found = [];
        bruteState.sub.list = [];
      }

      bruteState.sub.running = true;

      let list = bruteState.sub.list;
      if(!list.length){
        list = ["www","mail","api","dev","staging","test","admin","portal","blog","shop","cdn","static","app","vpn","secure","img","upload","files","backup","beta"];
        bruteState.sub.list = list;
      }

      while(bruteState.sub.running && bruteState.sub.index < list.length && bruteState.sub.index < 200){
        const s = list[bruteState.sub.index];
        bruteState.sub.index++;

        try{
          const r = await timeoutFetch(PROXY+"http://"+s+"."+domain, 3000);
          if(r && r.status === 200){
            bruteState.sub.found.push(s+"."+domain);
          }
        }catch(e){}

        const output =
          `Subdomain Brute Result:
` +
          `Found: ${bruteState.sub.found.length ? escapeHTML(bruteState.sub.found.join("
")) : "None"}
` +
          `Checked: ${bruteState.sub.index}/${Math.min(list.length, 200)}`;

        document.getElementById("subOut").innerHTML = output;
        await new Promise(r=>setTimeout(r, 150));
      }
    }

    // ----------------- WP DETECT -----------------
    async function wpPluginThemeDetect(){
      const domain = clean(document.getElementById("target").value);
      if(!domain){
        document.getElementById("subOut").innerHTML="Target required.";
        return;
      }

      const url = PROXY+"http://"+domain;
      try{
        const res = await timeoutFetch(url, 8000);
        const body = await res.text();

        const plugins = [...body.matchAll(/wp-content\/plugins\/([^\/]+)\//gi)].map(x=>x[1]);
        const themes = [...body.matchAll(/wp-content\/themes\/([^\/]+)\//gi)].map(x=>x[1]);

        const puniq = [...new Set(plugins)];
        const tuniq = [...new Set(themes)];

        document.getElementById("subOut").innerHTML =
          `WP Plugin/Theme Detection:
` +
          `Plugins: ${puniq.length ? escapeHTML(puniq.join(", ")) : "None"}
` +
          `Themes: ${tuniq.length ? escapeHTML(tuniq.join(", ")) : "None"}

` +
          `Note: Only works if WP assets are exposed in HTML.`;
      }catch(e){
        document.getElementById("subOut").innerHTML="Failed (CORS/proxy/timeout)";
      }
    }
