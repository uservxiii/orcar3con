async function adminBrute(resume=false){
  const domain = clean(document.getElementById("target").value);
  if(!domain){
    document.getElementById("adminOut").textContent="Target required.";
    return;
  }
  if(!resume){
    bruteState.admin.running = true;
    bruteState.admin.index = 0;
    bruteState.admin.found = [];
    bruteState.admin.list = [];
  }
  bruteState.admin.running = true;
  let list = bruteState.admin.list;
  if(!list.length){
    const defaultList = [
      "admin","administrator","admin/login","admin/index","adminpanel","adminarea","cpanel",
      "wp-admin","wp-login.php","administrator/index.php","user/login","login","dashboard",
      "backend","admin1","admin2","manager","login.php","admin.php","admin/login.php",
      "adminpanel.php","adminarea.php","admin/home","admin/dashboard","admin/login",
      "panel","sysadmin","portal","adminconsole","console","webadmin","siteadmin",
      "wp-admin.php","wp-login.php"
    ];
    list = [...defaultList];
    const fileInput = document.getElementById("adminWordlistFile");
    if(fileInput.files.length){
      const file = fileInput.files[0];
      const text = await file.text();
      const lines = text.split("\n").map(x=>x.trim()).filter(Boolean);
      list = [...new Set([...list, ...lines])];
    }
    bruteState.admin.list = list;
  }
  while(bruteState.admin.running && bruteState.admin.index < list.length){
    const p = list[bruteState.admin.index];
    bruteState.admin.index++;
    try{
      const r = await timeoutFetch(PROXY+"http://"+domain+"/"+p, 4000);
      if(r.status===200){
        bruteState.admin.found.push("/"+p);
      }
    }catch{}
    document.getElementById("adminOut").textContent =
      `Admin Brute Result:\nFound: ${bruteState.admin.found.length ? bruteState.admin.found.join(", ") : "None"}\nChecked: ${bruteState.admin.index}/${list.length}`;
    await new Promise(r=>setTimeout(r, 150));
  }
  if(!bruteState.admin.running){
    document.getElementById("adminOut").textContent += "\n\nPaused. Press Resume to continue.";
  }
}
function csrfChecker(){
  const body = lastReport?.http?.body || "";
  const forms = [...body.matchAll(//gi)];
  if(!forms.length){
    document.getElementById("adminOut").textContent="No forms found on homepage.";
    return;
  }
  let issues = [];
  forms.forEach((f, idx)=>{
    const formHtml = f[0];
    const hasCSRF = /csrf|token|_csrf|csrf_token/i.test(formHtml);
    if(!hasCSRF){
      issues.push(`Form ${idx+1}: MISSING CSRF token`);
    }
  });
  document.getElementById("adminOut").textContent =
    `CSRF Token Checker:\n${issues.length ? issues.join("\n") : "All forms contain CSRF token (or token-like field)."}\n\n(Note: token detection is heuristic-based.)`;
}
async function subdomainBrute(resume=false){
  const domain = clean(document.getElementById("target").value);
  if(!domain){
    document.getElementById("subOut").textContent="Target required.";
    return;
  }
  if(!resume){
    bruteState.sub.running = true;
    bruteState.sub.index = 0;
    bruteState.sub.found = [];
    bruteState.sub.list = [];
  }
  bruteState.sub.running = true;
  let list = bruteState.sub.list;
  if(!list.length){
    list = ["www","mail","api","dev","staging","test","admin","portal","blog","shop","cdn","static","app","vpn","secure","img","upload","files","backup","beta"];
    const fileInput = document.getElementById("subWordlistFile");
    if(fileInput.files.length){
      const file = fileInput.files[0];
      const text = await file.text();
      const lines = text.split("\n").map(x=>x.trim()).filter(Boolean);
      list = [...new Set([...list, ...lines])];
    }
    bruteState.sub.list = list;
  }
  while(bruteState.sub.running && bruteState.sub.index < list.length && bruteState.sub.index < 200){
    const s = list[bruteState.sub.index];
    bruteState.sub.index++;
    try{
      const r = await timeoutFetch(PROXY+"http://"+s+"."+domain, 3000);
      if(r.status===200){
        bruteState.sub.found.push(s+"."+domain);
      }
    }catch{}
    document.getElementById("subOut").textContent =
      `Subdomain Brute Result:\nFound: ${bruteState.sub.found.length ? bruteState.sub.found.join("\n") : "None"}\nChecked: ${bruteState.sub.index}/${Math.min(list.length, 200)}`;
    await new Promise(r=>setTimeout(r, 150));
  }
  if(!bruteState.sub.running){
    document.getElementById("subOut").textContent += "\n\nPaused. Press Resume to continue.";
  }
}
async function wpPluginThemeDetect(){
  const domain = clean(document.getElementById("target").value);
  if(!domain){
    document.getElementById("subOut").textContent="Target required.";
    return;
  }
  const url = PROXY+"http://"+domain;
  try{
    const res = await timeoutFetch(url, 8000);
    const body = await res.text();
    const plugins = [...body.matchAll(/wp-content\/plugins\/([^\/]+)\//gi)].map(x=>x[1]);
    const themes = [...body.matchAll(/wp-content\/themes\/([^\/]+)\//gi)].map(x=>x[1]);
    const puniq = [...new Set(plugins)];
    const tuniq = [...new Set(themes)];
    document.getElementById("subOut").textContent =
      `WP Plugin/Theme Detection:\nPlugins: ${puniq.length ? puniq.join(", ") : "None"}\nThemes: ${tuniq.length ? tuniq.join(", ") : "None"}\n\nNote: Only works if WP assets are exposed in HTML.`;
  }catch(e){
    document.getElementById("subOut").textContent="Failed (CORS/proxy/timeout)";
  }
}

  bruteState[type].running=true;
